// Generación del cartel Canvas con salto de línea automático y ajustes de alineación async function crearCartelDinamico(pData, pctDescuento) { const canvas = document.createElement('canvas'); canvas.width = 800; canvas.height = 980; const ctx = canvas.getContext('2d'); const [imgProducto, imgMp, imgCarrefour] = await Promise.all([ cargarImagen(pData.fotoUrl), cargarImagen(URL_MP_LOGO), cargarImagen(URL_CARREFOUR_LOGO) ]); // Fondo amarillo ctx.fillStyle = "#ffe600"; ctx.fillRect(0, 0, canvas.width, canvas.height); // Header azul ctx.fillStyle = "#001a72"; ctx.font = "bold 38px sans-serif"; ctx.textAlign = "center"; ctx.fillText("HOY JUEVES CON", canvas.width / 2, 55); if (imgMp) { const w = 110, h = (imgMp.height / imgMp.width) * w; ctx.drawImage(imgMp, (canvas.width / 2) - (w / 2), 65, w, h); } // Recuadro blanco ctx.fillStyle = "#ffffff"; ctx.lineWidth = 4; ctx.strokeStyle = "#00a3e0"; ctx.beginPath(); ctx.roundRect(50, 135, 700, 680, 25); ctx.fill(); ctx.stroke(); if (imgCarrefour) { const w = 90, h = (imgCarrefour.height / imgCarrefour.width) * w; ctx.drawImage(imgCarrefour, 630, 150, w, h); } // Imagen del producto if (imgProducto) { const imgMaxW = 460, imgMaxH = 320; let scale = Math.min(imgMaxW / imgProducto.width, imgMaxH / imgProducto.height); let w = imgProducto.width * scale, h = imgProducto.height * scale; ctx.drawImage(imgProducto, (canvas.width - w) / 2, 310 - (h / 2), w, h); } if (pData.aclaracion) { ctx.fillStyle = "#000000"; ctx.font = "bold 20px sans-serif"; ctx.textAlign = "center"; ctx.fillText(pData.aclaracion.toUpperCase(), canvas.width / 2, 500); } // TEXTO DEL PRODUCTO (Alineado a la altura del cartel rojo) ctx.fillStyle = "#000000"; ctx.textAlign = "left"; ctx.font = "bold 24px sans-serif"; const palabras = pData.producto.split(" "); let linea = "", y = 570; // Borde superior alineado con el cuadro rojo (y=535) const maxAncho = 310; for (let n = 0; n < palabras.length; n++) { let testLinea = linea + palabras[n] + " "; if (ctx.measureText(testLinea).width > maxAncho && n > 0) { ctx.fillText(linea, 75, y); linea = palabras[n] + " "; y += 30; // Salto de línea } else { linea = testLinea; } } ctx.fillText(linea, 75, y); // Cartel Rojo y Precios sin separadores de miles ($2700) const posXCartel = 400, anchoCartel = 320; const centroCartel = posXCartel + (anchoCartel / 2); const precioFinal = Math.round(pData.precioRegular * (1 - pctDescuento / 100)); const precioReg = Math.round(pData.precioRegular); ctx.fillStyle = "#e11d48"; ctx.fillRect(posXCartel, 535, anchoCartel, 140); ctx.fillStyle = "#ffffff"; ctx.font = "bold 78px sans-serif"; ctx.textAlign = "center"; ctx.fillText("$" + precioFinal, centroCartel, 635); ctx.fillStyle = "#000000"; ctx.font = "bold 20px sans-serif"; ctx.fillText("PRECIO REGULAR $" + precioReg, centroCartel, 710); // Pie de página ctx.font = "bold 22px sans-serif"; ctx.textAlign = "center"; ctx.fillText("Abonando con QR desde el celular,", canvas.width / 2, 875); ctx.fillText("EXCLUSIVO DINERO EN CUENTA.", canvas.width / 2, 902); ctx.fillText("COMPRA MINIMA DE $15000", canvas.width / 2, 929); return canvas.toDataURL('image/png'); }
